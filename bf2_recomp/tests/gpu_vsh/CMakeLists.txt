cmake_minimum_required(VERSION 3.20)
project(bf2_gpu_vsh C)
set(CMAKE_C_STANDARD 11)

set(game ../..)
set(kernel ../../../tools/xboxrecomp/src/kernel)

file(READ "${game}/src/nv2a_vsh_gpu.hlsl" shader_code)
file(READ "${game}/src/nv2a_vsh_gather.hlsli" gather_code)
string(APPEND shader_code "\n${gather_code}")
string(REPLACE "\\" "\\\\" shader_code "${shader_code}")
string(REPLACE "\"" "\\\"" shader_code "${shader_code}")
string(REPLACE "\r" "" shader_code "${shader_code}")
string(REPLACE "\n" "\\n\"\n\"" BF2_VSH_SOURCE "${shader_code}")
configure_file("${game}/src/gpu_vsh_shader.h.in" gpu_vsh_shader.h @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${game}/src/nv2a_vsh_gpu.hlsl" "${game}/src/nv2a_vsh_gather.hlsli")

add_executable(gpu-vsh-test check.c "${kernel}/nv2a_vsh_interp.c")
target_include_directories(gpu-vsh-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${kernel}")
target_link_libraries(gpu-vsh-test PRIVATE d3d11 d3dcompiler dxgi dxguid)
target_compile_definitions(gpu-vsh-test PRIVATE _CRT_SECURE_NO_WARNINGS)
enable_testing()
add_test(NAME gpu-vsh-regression COMMAND gpu-vsh-test)
add_executable(gpu-gather-test gather.c "${kernel}/nv2a_vsh_interp.c")
target_include_directories(gpu-gather-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${kernel}")
target_link_libraries(gpu-gather-test PRIVATE d3d11 d3dcompiler dxgi dxguid)
target_compile_definitions(gpu-gather-test PRIVATE _CRT_SECURE_NO_WARNINGS)
add_test(NAME gpu-gather-regression COMMAND gpu-gather-test)
